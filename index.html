CONFIGURATION & TYPES // ============================================================================ const CONFIG = { PORT: process.env.PORT || 3000, OPENAI_API_KEY: process.env.OPENAI_API_KEY || "", ELEVENLABS_API_KEY: process.env.ELEVENLABS_API_KEY || "", REPLICATE_API_TOKEN: process.env.REPLICATE_API_TOKEN || "", DEFAULT_VOICE_ID: "21m00Tcm4TlvDq8ikWAM", // Rachel / Multilingual }; interface WordAlignment { word: string; start: number; end: number; } interface Scene { id: string; sceneNumber: number; voiceoverText: string; visualPrompt: string; cameraMovement: "zoom_in" | "zoom_out" | "pan_right" | "static"; imageUrl: string; audioBase64: string; durationInSec: number; alignmentData: WordAlignment[]; } interface Project { id: string; title: string; rawStory: string; genre: string; visualStyle: string; status: "PENDING" | "SCRIPTING" | "GENERATING" | "READY" | "FAILED"; progress: number; statusMessage: string; scenes: Scene[]; } // Mémoire locale (remplace la base de données pour une exécution autonome) const db = new Map(); const sseClients = new Map(); function notifyClients(projectId: string, data: object) { const clients = sseClients.get(projectId) || []; const payload = `data: ${JSON.stringify(data)}\n\n`; clients.forEach((res) => res.write(payload)); } // ============================================================================ // 2. PIPELINE D'ORCHESTRATION IA // ============================================================================ // Étape A : Découpage scénaristique (OpenAI) async function generateStoryboard(story: string, genre: string, visualStyle: string) { const res = await fetch("https://api.openai.com/v1/chat/completions", { method: "POST", headers: { "Content-Type": "application/json", Authorization: `Bearer ${CONFIG.OPENAI_API_KEY}`, }, body: JSON.stringify({ model: "gpt-4o", response_format: { type: "json_object" }, messages: [ { role: "system", content: `Tu es un réalisateur d'animation. Découpe l'histoire en scènes (3 à 6 scènes). Format JSON strict attendu: { "title": "Titre", "style_guide": "Description visuelle courte pour cohérence", "scenes": [ { "scene_number": 1, "voiceover_text": "Texte court à prononcer", "visual_prompt": "Prompt d'image détaillé", "camera_movement": "zoom_in" } ] } Style: ${visualStyle}, Genre: ${genre}.`, }, { role: "user", content: story }, ], }), }); const json = await res.json(); return JSON.parse(json.choices[0].message.content); } // Étape B : Audio TTS & Timestamps karaoké (ElevenLabs) async function generateAudio(text: string, voiceId: string) { const res = await fetch(`https://api.elevenlabs.io/v1/text-to-speech/${voiceId}/with-timestamps`, { method: "POST", headers: { "Content-Type": "application/json", "xi-api-key": CONFIG.ELEVENLABS_API_KEY, }, body: JSON.stringify({ text, model_id: "eleven_multilingual_v2", }), }); const data = await res.json(); const words: WordAlignment[] = []; const starts = data.alignment.character_start_times_seconds; const ends = data.alignment.character_end_times_seconds; const chars = data.alignment.characters; let word = ""; let wStart = 0; chars.forEach((c: string, i: number) => { if (word === "") wStart = starts[i]; if (c === " " || i === chars.length - 1) { if (c !== " ") word += c; if (word.trim()) words.push({ word: word.trim(), start: wStart, end: ends[i] }); word = ""; } else { word += c; } }); const duration = words.length ? words[words.length - 1].end + 0.3 : 3.0; return { audioBase64: data.audio_base64, words, duration }; } // Étape C : Génération d'images (Replicate / Flux) async function generateImage(prompt: string, styleGuide: string): Promise { const startRes = await fetch("https://api.replicate.com/v1/models/black-forest-labs/flux-schnell/predictions", { method: "POST", headers: { "Content-Type": "application/json", Authorization: `Bearer ${CONFIG.REPLICATE_API_TOKEN}`, }, body: JSON.stringify({ input: { prompt: `${prompt}, ${styleGuide}, ultra-detailed, 8k, cinematic`, aspect_ratio: "16:9", }, }), }); let prediction = await startRes.json(); while (prediction.status !== "succeeded" && prediction.status !== "failed") { await new Promise((r) => setTimeout(r, 1000)); const poll = await fetch(prediction.urls.get, { headers: { Authorization: `Bearer ${CONFIG.REPLICATE_API_TOKEN}` }, }); prediction = await poll.json(); } return prediction.output[0]; } // Orchestrateur en tâche de fond async function runPipeline(projectId: string) { const project = db.get(projectId)!; try { project.status = "SCRIPTING"; project.progress = 15; project.statusMessage = "Scénarisation avec GPT-4o..."; notifyClients(projectId, project); const storyboard = await generateStoryboard(project.rawStory, project.genre, project.visualStyle); project.title = storyboard.title; project.status = "GENERATING"; const total = storyboard.scenes.length; for (let i = 0; i < total; i++) { const s = storyboard.scenes[i]; project.progress = 20 + Math.round(((i + 1) / total) * 75); project.statusMessage = `Création des visuels et voix (Scène ${i + 1}/${total})...`; notifyClients(projectId, project); const [audioRes, imgUrl] = await Promise.all([ generateAudio(s.voiceover_text, CONFIG.DEFAULT_VOICE_ID), generateImage(s.visual_prompt, storyboard.style_guide), ]); project.scenes.push({ id: `sc_${i + 1}`, sceneNumber: s.scene_number, voiceoverText: s.voiceover_text, visualPrompt: s.visual_prompt, cameraMovement: s.camera_movement || "zoom_in", imageUrl: imgUrl, audioBase64: audioRes.audioBase64, durationInSec: audioRes.duration, alignmentData: audioRes.words, }); } project.status = "READY"; project.progress = 100; project.statusMessage = "Génération terminée !"; notifyClients(projectId, project); } catch (err: any) { project.status = "FAILED"; project.statusMessage = err.message || "Erreur durant la génération."; notifyClients(projectId, project); } } // ============================================================================ // 3. SERVEUR HTTP REST + ROUTEUR SSE // ============================================================================ const server = http.createServer(async (req, res) => { const parsedUrl = new URL(req.url || "/", `http://${req.headers.host}`); const pathname = parsedUrl.pathname; // Servir l'interface web if (req.method === "GET" && pathname === "/") { res.writeHead(200, { "Content-Type": "text/html; charset=utf-8" }); return res.end(FRONTEND_HTML); } // API : Créer un projet if (req.method === "POST" && pathname === "/api/projects") { let body = ""; req.on("data", (chunk) => (body += chunk)); req.on("end", async () => { const { story, genre, visualStyle } = JSON.parse(body); const projectId = "proj_" + Date.now(); const newProject: Project = { id: projectId, title: "Nouvelle Histoire", rawStory: story, genre, visualStyle, status: "PENDING", progress: 0, statusMessage: "En attente...", scenes: [], }; db.set(projectId, newProject); runPipeline(projectId); // Traitement asynchrone non-bloquant res.writeHead(201, { "Content-Type": "application/json" }); res.end(JSON.stringify({ projectId })); }); return; } // API : Server-Sent Events pour la progression temps réel if (req.method === "GET" && pathname.startsWith("/api/sse/")) { const projectId = pathname.split("/").pop()!; res.writeHead(200, { "Content-Type": "text/event-stream", "Cache-Control": "no-cache", Connection: "keep-alive", }); if (!sseClients.has(projectId)) sseClients.set(projectId, []); sseClients.get(projectId)!.push(res); if (db.has(projectId)) { res.write(`data: ${JSON.stringify(db.get(projectId))}\n\n`); } req.on("close", () => { const clients = sseClients.get(projectId) || []; sseClients.set(projectId, clients.filter((c) => c !== res)); }); return; } res.writeHead(404); res.end("Not Found"); }); // ============================================================================ // 5.
